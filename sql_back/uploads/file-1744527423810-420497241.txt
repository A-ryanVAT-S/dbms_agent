SELECT
    p.product_id,
    p.name AS product_name,
    c.name AS category_name,
    p.stock_quantity,
    p.price,
    -- Oracle's equivalent for DATE()
    TRUNC(p.launch_date) AS launch_day,
    -- Oracle's equivalent for string concatenation and formatting
    p.name || ' - $' || TO_CHAR(p.price, '9999999.99') AS formatted_product_price
FROM
    products p
INNER JOIN
    categories c ON p.category_id = c.category_id
WHERE
    c.name = 'Electronics'
    AND p.is_active = 1 -- Oracle also supports 1 for TRUE
    AND p.launch_date >= DATE '2023-01-01' -- Oracle's date literal
ORDER BY
    p.price DESC,
    p.name ASC
FETCH FIRST 10 ROWS ONLY; -- Oracle's equivalent for LIMIT
